\section{Dictionary between Measure Theory and Probability}%
We will assume a lot of measure theory background, so we will instead discuss the dictionary between terms from probability and terms from measure theory. It's important to note that the measure spaces in probability theory are probability measure spaces (i.e., they have measure $1$).
\begin{center}
  \renewcommand{\arraystretch}{1.5}
  \begin{tabular}{c|c}
    Measure Theory & Probability\\
    \hline
    Set $X$ & Sample space $\Omega$\\
    $\sigma$-Algebra $\Sigma$ & $\sigma$-Field $\mathcal{F}$\\
    Measurable subsets $A\in \Sigma$ & Events $F\in \mathcal{F}$\\
    Measure $\mu(A)$ & Probability $P(F)$\\
    Almost Everywhere (a.e.) & Almost Surely (a.s.)\\
    $\ds \limsup_{n\rightarrow\infty} E_n$ & $\set{E_n\text{ infinitely often}}$, $\set{E_n\text{ i.o.}}$\\
    $\ds \liminf_{n\rightarrow\infty} E_n$ & $\set{E_n\text{ eventually}}$, $ \set{E_n\text{ ev}} $\\
    Borel Measurable Function $f\colon X\rightarrow \R$ & Random Variable $X\colon \Omega\rightarrow \R$\\
    Integral $ \int_{}^{} f\:d\mu $ & Expectation $E[X]$\\
    Pushforward measure $f_{\ast}\mu$ & Law $\Lambda_X$\\
    $\mu\left( S \right)$ & $P\left( \omega\in S \right)$\\
    $f_{\ast}\mu\left( S' \right)$ & $P\left( X\in S' \right)$\\
    \hline
  \end{tabular}
\end{center}
\begin{remark}
  In probability, rather than the non-calligraphic $L^p$ spaces (whose elements are equivalence classes of almost-everywhere equal functions), one uses the calligraphic $\mathcal{L}^p$, whose elements are the functions themselves. This does not fundamentally change anything in particular (as probabilists are only concerned with almost-everywhere equivalence regardless), but it is useful to note.
\end{remark}
\subsection{Important Theorems}%
\subsubsection{$\sigma$-Algebras and Measures}%
\begin{definition}
  Let $S$ be a set, and let $\mathcal{D}$ be a collection of subsets. We call $ \mathcal{D} $ a $\pi$-system on $S$ if it is closed under finite intersection.

  We call $\mathcal{D}$ a $\lambda$-system if
  \begin{itemize}
    \item $S\in \mathcal{D}$;
    \item for any $A,B\in \mathcal{D}$ with $A\subseteq B$, one has $B\setminus A\in \mathcal{D}$;
    \item if $\left( A_n \right)_n\subseteq \mathcal{D}$ is such that $\bigcup_n A_n = A$, then $A\in \mathcal{D}$.
  \end{itemize}
  For an arbitrary collection of subsets $\mathcal{I}$, we write $\pi\left( \mathcal{I} \right)$ for the $\pi$-system generated by $ \mathcal{I} $ and $\lambda\left( \mathcal{I} \right)$ for the $\lambda$-system generated by $ \mathcal{I} $.
\end{definition}
\begin{proposition}
  A collection $\Sigma$ of subsets of $S$ is a $\sigma$-algebra if and only if $\Sigma$ is both a $\pi$-system and a $\lambda$-system.
\end{proposition}
\begin{proof}
  Let $\Sigma$ be a $\pi$-system and a $\lambda$-system. If $E,F\in \Sigma, \left( E_n \right)_n\subseteq \Sigma$, then 
  \begin{align*}
    E^{c} = S\setminus E\\
    E\cup F &= S\setminus \left( E^{c}\cap F^{c} \right).
  \end{align*}
  Therefore, $G_n = E_1\cup\cdots\cup E_n\in \Sigma$, with $\bigcup_n G_n = \bigcup_k E_k$, so that $\bigcup_{k} E_k\in \Sigma$. Finally, we have $\bigcap_{k} E_k = \left( \bigcup_{k}E_k^{c} \right)^{c}$, meaning that $\Sigma$ is a $\sigma$-algebra.
\end{proof}
\begin{theorem}[Dynkin $\pi$-$\lambda$ Theorem]
  If $ \mathcal{I} $ is a $\pi$-system, then $\lambda\left( \mathcal{I} \right) = \sigma\left( \mathcal{I} \right)$.
\end{theorem}
\begin{proof}
  To prove the theorem, we will use a classic technique by defining a ``truth set'' on which a desired statement holds, then showing that truth set is the desired $\sigma$-algebra.

  Define
  \begin{align*}
    \mathcal{D}_1 &= \set{B\in \lambda\left( \mathcal{I} \right) : B\cap C\in \lambda\left( \mathcal{I} \right)\text{  for all }C\in \mathcal{I}}.
  \end{align*}
  We observe that $\mathcal{I}$ is a $\pi$ system, so that $\mathcal{I}\subseteq \mathcal{D}_1$. Furthermore, 
  \begin{itemize}
    \item $S\in \mathcal{D}_1$;
    \item for any $B_1,B_2\in \mathcal{D}_1$ with $B_1\subseteq B_2 $, we have that for any given $C\in \mathcal{I}$, $\left( B_2\setminus B_1 \right)\cap C = \left( B_2\cap C \right)\setminus \left( B_1\cap C \right)$, so since $\lambda\left( \mathcal{I} \right)$ is a $\lambda$-system, it follows that $B_2\setminus B_1\in \mathcal{D}_1$;
    \item if $\left( B_n \right)_n\subseteq \mathcal{D}_1$ increases to $B$, then for each $C\in \mathcal{I}$, one has $\left( B_n\cap C \right)_n\nearrow B\cap C$, so that $B\cap C\in \lambda\left( \mathcal{I} \right)$, meaning $B\in \mathcal{D}_1$.
  \end{itemize}
  Thus, $\mathcal{D}_1$ is a $\lambda$-system containing $I$, so since $\mathcal{D}_1\subseteq \lambda\left( \mathcal{I} \right)$ by definition, one has $\mathcal{D}_1 = \lambda\left( \mathcal{I} \right)$.

  Next, define
  \begin{align*}
    \mathcal{D}_2 &= \set{A\in \lambda\left( \mathcal{I} \right) : B\cap A\in \lambda\left( \mathcal{I} \right)\text{ for all }B\in \lambda\left( \mathcal{I} \right)}.
  \end{align*}
  From above, we have that $\mathcal{D}_2$ contains $I$. Similarly to above, we can prove that $\mathcal{D}_2$ inherits the $\lambda$-system structure from $\lambda\left( \mathcal{I} \right)$, so that $\mathcal{D}_2 = \lambda\left( \mathcal{I} \right)$. Yet, this means that $\mathcal{D}_2 = \lambda\left( \mathcal{I} \right)$, so $\lambda\left( \mathcal{I} \right)$ is a $\pi$-system.
\end{proof}
\begin{corollary}
  If two probability measures $\mu_1$ and $\mu_2$ agree on a $\pi$-system $\mathcal{I}$, then they agree on the $\sigma$-algebra $\sigma\left( \mathcal{I} \right)$.
\end{corollary}
\begin{theorem}[Carathéodory Extension Theorem]
  Let $S$ be a set, $\Sigma_0$ an algebra of subsets on $S$, and $\Sigma = \sigma\left( \Sigma_0 \right)$. If $\mu_0\colon \Sigma_0\rightarrow [0,\infty]$ is a countably additive map, then there exists a measure $\mu$ on $\left( S,\Sigma \right)$ such that $\mu = \mu_0$ on $\Sigma_0$.

  Furthermore, if $\mu_0\left( S \right) < \infty$, then the extension is unique.
\end{theorem}
\subsection{Convergence Theorems}%
We recall some of the useful convergence theorems from measure theory, specializing to the case with measure $1$ and rewriting in the language of probability.
\begin{theorem}[Monotone Convergence Theorem]
  If $\left( X_n \right)_n$ is a sequence of real-valued random variables increasing pointwise almost surely to a real-valued random variable $X$, then
  \begin{align*}
    E[X] &= \lim_{n\rightarrow\infty} E\left[ X_n \right].
  \end{align*}
\end{theorem}
\begin{theorem}[Fatou's Lemma]
  If $\left( X_n \right)_n$ is a sequence of positive-valued random variables, then
  \begin{align*}
    E\left[ \liminf_{n\rightarrow\infty}X_n \right] &\leq \liminf_{n\rightarrow\infty} E\left[ X_n \right]\\
    \limsup_{n\rightarrow\infty} E\left[ X_n \right] &\leq E\left[ \limsup_{n\rightarrow\infty}E\left[ X_n \right] \right].
  \end{align*}
\end{theorem}
\begin{theorem}[Dominated Convergence Theorem]
  Let $\left( X_n \right)_n$ be a sequence of complex-valued random variables converging pointwise almost surely to a complex-valued random variable $X$. If there exists some positive-valued integrable random variable $Y$ such that $\left\vert X_n \right\vert\leq Y$ for all $n$, then $\left( X_n \right)_n\xrightarrow{\mathcal{L}^1}X$. That is,
  \begin{align*}
    E\left[ \left\vert X_n-X \right\vert \right] &\rightarrow 0,
  \end{align*}
  so that $E\left[ X_n \right]\rightarrow E\left[ X \right]$.
\end{theorem}
\begin{theorem}[Scheffé Convergence Theorem]
  If $\left( X_n \right)_n$ is a sequence of (real valued) integrable random variables converging to an integrable random variable $X$ almost surely, and $E\left( \left\vert X_n \right\vert \right)\rightarrow E\left( \left\vert X \right\vert \right)$, then $\left( X_n \right)_n\xrightarrow{\mathcal{L}^1}X$.
\end{theorem}
\begin{remark}
  The Scheffé convergence theorem is proven by splitting into positive and negative components.
\end{remark}
\subsection{Independence}%
\begin{definition}
  We say sub-$\sigma$-algebras $\mathcal{G}_1,\mathcal{G}_2,\dots$ of $\mathcal{F}$ are \textit{independent} if for any $G_i\in \mathcal{G}_i$ with distinct $i$, one has
  \begin{align*}
    P\left( G_{i_1}\cap\cdots\cap G_{i_n} \right) &= \prod_{k=1}^{n} P\left( G_{i_k} \right).
  \end{align*}
  We say random variables $X_1,X_2,\dots$ are independent if their produced $\sigma$-algebras $\sigma\left( X_1 \right),\sigma\left( X_2 \right),\dots$ are independent.

  Events $E_1,E_2,\dots$ are independent if their $\sigma$-algebras $\mathcal{E}_n$, where
  \begin{align*}
    \mathcal{E}_n &= \set{\emptyset,E_n,E_n^{c},\Omega},
  \end{align*}
  are independent. 
\end{definition}
\begin{theorem}
  If $X$ and $Y$ are independent random variables, then
  \begin{align*}
    E\left[ XY \right] &= E\left[ X \right]E\left[ Y \right].
  \end{align*}
\end{theorem}
\begin{proof}
  Bootstrap from simple functions.
\end{proof}
\subsection{Asymptotic Results}%
\begin{theorem}[First Borel--Cantelli Lemma]
  Let $\left( E_n \right)_n$ be a family of events such that $\sum_{n}P\left( E_n \right) < \infty$. Then,
  \begin{align*}
    P\left( \limsup_{n\to\infty} E_n \right) &= P\left( E_n\text{ i.o.} \right)\\
                                             &= 0.
  \end{align*}
\end{theorem}
\begin{proof}
  Let
  \begin{align*}
    G_m &= \bigcup_{n=m}^{\infty} E_n.
  \end{align*}
  Then, $G_m\searrow \limsup E_n$, and one has
  \begin{align*}
    P\left( \limsup E_n \right) &\leq P\left( G_m \right)\\
                                &\leq \sum_{n=m}^{\infty}P\left( E_n \right).
  \end{align*}
  Taking $m\rightarrow\infty$ yields our desired result.
\end{proof}
\begin{theorem}[Second Borel--Cantelli Lemma]
  Suppose $\left( E_n \right)_n$ is a sequence of independent events. Then, if $\sum_{n}P\left( E_n \right) = \infty$, one has
  \begin{align*}
    P\left( \limsup E_n \right) &= 1.
  \end{align*}
\end{theorem}
\begin{proof}
  Recall that $\left( \limsup E_n \right)^{c} = \liminf E_n^{c}$. If $p_n = P\left( E_n \right)$, then one has
  \begin{align*}
    P\left( \bigcap_{n\geq m}E_n^{c} \right) &= \prod_{n\geq m} \left( 1-p_n \right).
  \end{align*}
  Notice that if $x \geq 0$, then $1-x \leq e^{-x}$. Therefore, we have
  \begin{align*}
    \prod_{n\geq m} \left( 1-p_n \right) &\leq \exp\left( -\sum_{n\geq m}p_n \right)\\
                                         &= 0,
  \end{align*}
  where the latter equality emerges from the fact that $\sum_{n}p_n = \infty$.

  Therefore $P\left( \liminf E_n^{c} \right) = 0$, meaning $P\left( \limsup E_n \right) = 1$.
\end{proof}
\subsection{Inequalities}%
\begin{theorem}[Chebyshev's Inequality]
  Let $Z$ be a measurable function on $\left( \Omega, \mathcal{F},P \right)$, and let $g\colon \R\to[0,\infty]$ be a Borel measurable increasing function. Then,
  \begin{align*}
    P\left( Z\geq c \right) &\leq \frac{1}{c}E\left[ Z \right]
  \end{align*}
  for all $c\in \R$.
\end{theorem}
\begin{proof}
  We observe that
  \begin{align*}
    E\left[ Z \right] &= \int_{}^{} Z\:dP\\
                      &\geq \int_{\set{Z\geq c}}^{} Z\:dP\\
                      &\geq \int_{\set{Z\geq c}}^{} \:dP\\
                      &= P\left( Z\geq c \right).
  \end{align*}
\end{proof}
\begin{theorem}[Jensen's Inequality]
  Let $c\colon G\rightarrow \R$ be a convex function on an open interval, and let $X$ be a random variable such that
  \begin{itemize}
    \item $E\left[ \left\vert X \right\vert \right] < \infty$;
    \item $P\left( X\in G \right) = 1$;
    \item $E\left( \left\vert c(X) \right\vert \right) < \infty$.
  \end{itemize}
  Then, $E\left[ c(X) \right]\geq c\left( E\left[ X \right] \right)$.
\end{theorem}
\begin{proof}
  Since $c$ is convex, we observe that whenever $u < v < w$, we have $\frac{c(v)-c(u)}{v-u}\leq \frac{c(w)-c(v)}{w-v}$. In particular, since for any $x <y$ in $G$ there are $z,w\in G$ with $z >w > y$, it follows that $c$ is continuous as we have $\left\vert c(y)-c(x) \right\vert\leq \frac{c(z)-c(w)}{z-w}\left\vert y-x \right\vert$.

  Therefore, the monotone derivatives
  \begin{align*}
    \left( D_{-}c \right)(v) &= \lim_{u\nearrow v} \frac{c(v)-c(u)}{v-u}\\
    \left( D_{+}c \right)(v) &= \lim_{w\searrow v} \frac{c(w)-c(v)}{w-v}
  \end{align*}
  exist and have $\left( D_{-}c \right)(v) \leq \left( D_{+}c \right)(v)$.

  These functions are increasing, and for every $v\in G$ and any $m\in \left[ \left( D_{-}c \right)(v),\left( D_{+}c \right)(v) \right]$, we have $c(x)\geq m\left( x-v \right) + c(v)$ for all $x\in G$.

  Therefore, if $\mu = E\left[ X \right]$, we have $c(X)\geq m\left( X-E[X] \right) + c\left( E[X] \right)$ for any $m\in \left[ \left( D_{-}c \right)\left(E[X]\right),\left( D_{+}c \right)\left( E[X] \right) \right]$.

  Taking expectations gives the desired result.
\end{proof}
\begin{theorem}[Hölder's Inequality]
  If $p$ and $q$ are conjugate exponents, then for any $f\in \mathcal{L}^p$ and $g\in \mathcal{L}^q$, we have $fg\in \mathcal{L}^1$ with $\norm{fg}_{\mathcal{L}^1}\leq \norm{f}_{\mathcal{L}^p}\norm{g}_{\mathcal{L}^q}$.
\end{theorem}
\begin{theorem}[Minkowski's Inequality]
  For any $f,g\in \mathcal{L}^p$, we have $\norm{f + g}_{\mathcal{L}^p}\leq \norm{f}_{\mathcal{L}^p} + \norm{g}_{\mathcal{L}^p}$.
\end{theorem}
\begin{theorem}[Inequality of $\mathcal{L}^p$ Norms]
  Suppose $1\leq p \leq r < \infty$, and $Y\in \mathcal{L}^r$. Then, $Y\in \mathcal{L}^p$, and we have $\norm{Y}_{\mathcal{L}^p}\leq \norm{Y}_{\mathcal{L}^r}$.
\end{theorem}
\begin{theorem}[Cauchy--Schwarz Inequality]
  If $X$ and $Y$ are in $\mathcal{L}^2$, then $XY\in \mathcal{L}^1$ and satisfy $E\left[ \left\vert XY \right\vert \right]\leq \norm{X}_{\mathcal{L}^2}\norm{Y}_{\mathcal{L}^2}$.
\end{theorem}
