\section{Motivation}%
Consider a space $V$ of (sufficiently nice) functions $f\colon \R\rightarrow \R$. Then, similar to how a vector in $\R^n$ can be denoted as
\begin{align*}
  x &= \begin{pmatrix}x_1\\\vdots\\x_n\end{pmatrix},
\end{align*}
we may view the function $f\colon \R\rightarrow \R$ as a ``column vector'' with uncountable index. That is, $f = \left( f(x) \right)_{x\in \R}$. If $L\colon V\rightarrow V$ is a linear map, then an analogue of the finite-dimensional matrix representation can be viewed as a kernel that $f$ is integrated against; that is,
\begin{align*}
  \left( Lf \right)(x) &= \int_{\R}^{} K\left( x,y \right)f(y)\:dy.
\end{align*}
In the special case that $L = \id_V$, then the kernel $K\left( x,y \right)$ is given by the Dirac delta distribution, $K\left( x,y \right) = \delta\left( x-y \right)$. The space of these distributions is the primary object of study in Fourier analysis.

\section{Locally Convex Topological Vector Spaces}%
Recall that if $X$ is a set, then $ \mathcal{T}\subseteq \mathcal{P}\left( \mathcal{P}\left( X \right) \right) $ is a topology on $X$ if it satisfies:
\begin{itemize}
  \item $\emptyset,X\in \mathcal{T}$;
  \item $ U_1,U_2\in \mathcal{T} $ implies $U_1\cap U_2\in \mathcal{T}$;
  \item $\set{U_i : i\in I}\subseteq \mathcal{T}$ implies $\bigcup_{i\in I}U_i\in \mathcal{T}$.
\end{itemize}
\begin{definition}
  \subsection{Definitions}%
  We say $\left( V, \mathcal{T}\right)$ is a \textit{topological vector space} (over $\K$) if $V$ is a $\K$-vector space whose topology is such that the maps
  \begin{align*}
    V\times V &\rightarrow V\\
    \left( u,v \right) &\mapsto u + v
      \intertext{and}
    \K\times V &\rightarrow V\\
    \left( \lambda,v \right) &\mapsto \lambda v
  \end{align*}
  are continuous with respect to the product topology. Note that $\K$ is equipped with its traditional topology.
\end{definition}
The morphisms of topological vector spaces are the \textit{continuous} linear maps. We say $V\cong W$ are isomorphic as topological vector spaces if there is $\varphi\colon V\rightarrow W$ that is bijective, linear, and continuous, and $\varphi^{-1}$ is also continuous.
\begin{definition}
  Let $V$ be a $\K$-vector space. We say $\rho\colon V\rightarrow [0,\infty)$ is a \textit{seminorm} if
  \begin{align*}
    \rho\left( v + w \right) &\leq \rho\left( v \right) + \rho\left( w \right)\\
    \rho\left( \lambda v \right) &= \left\vert \lambda \right\vert \rho(v)
  \end{align*}
  for any $v,w\in V$ and $\lambda\in \K$.
\end{definition}
\begin{remark}
  The only difference between a seminorm and a norm is that $\rho(v) = 0$ need not imply that $v = 0$. A norm is a seminorm.
\end{remark}
We will let $ \operatorname{SN}(V) $ be the set of all seminorms of $V$. 
\begin{definition}
  Let $V$ be a $\K$-vector space. Let $ \mathcal{N}\subseteq \operatorname{SN}(V) $. Define $ \mathcal{T} \left( \mathcal{N} \right) $ to be a topology with basic open sets $U\left( x,F,\ve \right)\subseteq V$ given by
  \begin{align*}
    U\left( x,F,\ve \right) &= \set{y\in V : \rho\left( y-x \right) < \ve\text{ for all }\rho\in F}\\
                            &= \bigcap_{\rho\in F} U\left( x,\rho,\ve \right).
  \end{align*}
  for any $x\in V$, $ F\subseteq \mathcal{N} $ finite, and $\ve > 0$.
\end{definition}
\begin{remark}
  I prefer to use the subbasic open sets
  \begin{align*}
    U\left( x,\rho, \ve \right)&= \set{y\in V : \rho\left( y-x \right) < \ve}
  \end{align*}
  and write intersections.
\end{remark}
We observe that the subsets of the form $U\left( x,\set{\rho},\ve \right)$ as $\rho$ ranges across $ \mathcal{N} $ forms a subbase for the topology of $V$.
\begin{proposition}
  If $F\subseteq \mathcal{N}$ is finite, $ \ve > 0 $, and $x\in V$, then
  \begin{align*}
    U\left( x,F,\ve \right)\in \mathcal{T}\left( \mathcal{N} \right).
  \end{align*}
\end{proposition}
\begin{proof}
  Suppose $y\in U\left( x,F,\ve \right)$. Our goal is to find $\eta > 0$ such that $U\left( y,F,\eta \right) \subseteq U\left( x,F,\ve \right)$. Suppose $z\in U\left( y,F,\eta \right)$. If $\rho\in F$, then
  \begin{align*}
    \rho\left( z-x \right) &\leq \rho\left( z-y \right) + \rho\left( y-x \right)\\
                           &< \eta + \rho\left( y-x \right).
  \end{align*}
  Therefore, we will define $\eta$ to be small enough such that $\eta + \max_{\rho\in F} \rho\left( y-x \right) < \ve$. This gives the desired result.
\end{proof}
\begin{definition}
  A topological space is called a locally convex topological vector space (LCTVS) if the topology is generated by a family of seminorms $\mathcal{N}$ with the property that
  \begin{align*}
    \bigcap_{\rho\in \mathcal{N}} \set{v : \rho(v) = 0} = \set{0}.
  \end{align*}
  A family $\mathcal{N}$ of seminorms that satisfies this property is said to be \textit{separating}.
\end{definition}
\begin{example}
  If $ H $ is a Hilbert space with orthonormal basis $\set{e_n : n\in \N}$, then the seminorms $ v\mapsto \left\vert \iprod{v}{e_i} \right\vert $ are separating.
\end{example}
%Theorem/Proof required
\begin{theorem}
  If $V$ is a $\K$-LCTVS and $ \mathcal{N} $ is a collection of seminorms that defines the topology on $V$, then the pair $ \left( V,\mathcal{T}\left( \mathcal{N} \right) \right) $ is Hausdorff if and only if $ \mathcal{N} $ separates the points of $V$.
\end{theorem}
\begin{proof}
  Assume $\mathcal{N}$ separates the points of $V$. Suppose $v\neq w$ are distinct elements of $V$. Then, since $v-w\neq 0$, there is some $\rho\in \mathcal{N}$ such that $\rho\left( v-w \right) = \ve_0 > 0$. Consider the finite set $F = \set{\rho}$, and choose some $0 < \ve < \ve_0/2$. Then, we claim that the basic open sets $U\left( v,F,\ve \right)$ and $U\left( w,F,\ve \right)$ are disjoint. This follows from the fact that if $x\in U\left( v,F,\ve \right)$, then $\ve_0 = \rho\left( v-w \right)\leq \rho\left( v-x \right) + \rho\left( x-w \right)$, so that $\rho\left( w-x \right) \geq \ve_0 - \ve > \ve_0 - \ve_0/2 > \ve$, yielding the desired result.

  Now, suppose $ \mathcal{N} $ does not separate the points of $V$. Then, there are some distinct $v\neq w$ such that for all $\rho\in \mathcal{N}$, we have $\rho\left( v-w \right) = 0$. In particular, observe that for all finite subsets $F\subseteq \mathcal{N}$ and all $\ve > 0$, we have that $w$ is contained in $U\left( v,F,\ve \right)$, meaning that there are no basic open sets that separate $v$ and $w$. Since any open subset of $V$ contains a basic open subset, it follows that for any open subsets $v\in U$ and $w\in U'$, we have some finite $F$ such that $U\left( v,F,\ve \right)\subseteq U$, but this also has $w\in U\left( v,F,\ve \right)$, implying that $U\cap U'\neq \emptyset$. Thus, $ V $ is not Hausdorff.
\end{proof}

\subsection{Examples}%
\begin{remark}
  If we remove the assumption that the family $\mathcal{N}$ is separating, then the topology induced by the trivial seminorm $\rho\colon V\rightarrow [0,\infty)$ is the indiscrete topology on $V$. Similarly, there is a family of seminorms that induces the discrete topology on $V$.
\end{remark}
\begin{enumerate}[(i)]
  \item Let $\left( V,\norm{\cdot} \right)$ be a normed vector space over $\K$. Examples of these include Banach spaces such as $L^p\left( X, \Sigma,\mu \right)$, where $\Sigma$ is a $\sigma$-algebra and $\mu$ is a positive measure on the measurable space $\left( X,\Sigma \right)$, and $p\in [1,\infty]$. 

    There is a metric on this normed space $d\left( x,y \right) = \norm{x-y}$, which gives rise to the metric topology $ \mathcal{T}\left( d \right) $. We have $U\in \mathcal{T}\left( d \right)$ if and only if $U\subseteq V$ is such that for any $x\in U$, there exists $\ve > 0$ such that
    \begin{align*}
      \set{y\in V : d\left( x,y \right) < \ve} &\eqcolon U\left( x,\ve \right)\\
                                               &\subseteq U.
    \end{align*}
    This is a locally convex space with seminorm $ \mathcal{N} = \set{\norm{\cdot}} $. There are two possibilities for finite subsets $F\subseteq \set{\norm{\cdot}}$, meaning that if $F = \emptyset$, then $U\left( x,F,\ve \right) = U\left( x,\emptyset,\ve \right) = V$, while if $F = \set{\norm{\cdot}}$, then
    \begin{align*}
      U\left( x,F,\ve \right) &= \set{y\in V : \rho\left( x-y \right) < \ve\text{ for all }\rho \in F}\\
                              &= \set{y\in V : \norm{x-y} < \ve}\\
                              &= U\left( x,\ve \right).
    \end{align*}
  \item Define $s \left( \N,\K \right)$ to be the space of sequences $x = \left( x_n \right)_{n\geq 0}$ in $\K^{\N}$ such that for all $ k\geq 0 $, $\sup_{n\geq 0} \left\langle n \right\rangle^{k} \left\vert x_n \right\vert < \infty$, where $ \left\langle n \right\rangle = \sqrt{1 + n^2} $ is the bracket. This is the space of \textit{rapidly decaying sequences} in $\K$. In essence, elements of $s\left( \N,\K \right)$ decay faster than any inverse power of the index $n$.

    We define the seminorms 
    \begin{align*}
      \rho_{k}(x) &= \sup_{n\geq 0} \left\langle n \right\rangle^{k} \left\vert x_n \right\vert.
    \end{align*}
    In fact, the $\set{\rho_k : k\in \N}$ are norms. Let
    \begin{align*}
      \mathcal{N}\coloneq \set{\rho_k : k\in \N}
    \end{align*}
    We obtain a locally convex on $s\left( \N,\K \right)$ with topology induced by the family $ \mathcal{N} $.

    Later, we will see that $ \left( s\left( \N,\K \right), \mathcal{T}\left( \mathcal{N} \right) \right) $ is \textit{not} a normed space. That is, there does not exist a norm $\norm{\cdot}$ such that $ \mathcal{T}\left( \mathcal{N} \right) = \mathcal{T}\left( \set{\norm{\cdot}} \right) $.
  \item Take $V = C\left( [0,\infty),\R \right)$ to be the set of all real-valued continuous functions on $[0,\infty)$ with pointwise addition and scalar multiplication. This is a real vector space.

    At first, we may think $\norm{f} = \sup_{t\in \R}\left\vert f(t) \right\vert$ would define a norm, but this is not satisfactory since the functions in $V$ are not necessarily bounded.

    Therefore, we want to restrict to compact subsets. Take
    \begin{align*}
      \rho_k(f) &\coloneq \sup_{0\leq t \leq k} \left\vert f(t) \right\vert.
    \end{align*}
    Notice that the $\rho_k$ are not norms (use Urysohn's Lemma or your favorite piecewise function to construct one). However, the family $\mathcal{N} = \set{\rho_k : k\geq 0}$ is a separating family.

    Consider the set of functions $V_0 = \set{f\in V : f(0) = 0}$. Then, $V_0\leq V$ is a closed subset.
    \begin{proof}
      Consider the linear functional $ \operatorname{ev}_0\colon V\rightarrow \R $ given by $f\mapsto f(0)$. We claim that this linear functional is continuous.

      Let $U\subseteq \R$ be open. Let $z\in U$, and let $\ve > 0$ be such that $U\left( z,\ve \right)\coloneq \set{y\in \R : \left\vert z-y \right\vert < \ve}$ is contained in $U$. Let $f\in \operatorname{ev}_0^{-1}\left( U\left( z,\ve \right) \right)$, and set $\ve_0 = \left\vert z - f(0) \right\vert$. To show that $ \operatorname{ev}_{0}^{-1}\left( U\left( z,\ve \right) \right) $ is open, it suffices to find some seminorm $\rho_k$ and some $\delta > 0$ such that $U\left( f,\rho_k,\delta \right)\subseteq \operatorname{ev}_0^{-1}\left( U\left( z,\ve \right) \right)$.

      In fact, we may choose $k = 1$ and $\delta = \ve - \ve_0$. Suppose $g\in U\left( f,\rho_1, \delta\right)$. Then, we have that 
      \begin{align*}
        \left\vert g(0) - z \right\vert &\leq \left\vert g(0) - f(0) \right\vert + \left\vert f(0) - z \right\vert\\
                                        &\leq \sup_{t\in [0,1]} \left\vert g(t)-f(t) \right\vert + \ve_0\\
                                        &< \delta + \ve_0\\
                                        &= \ve.
      \end{align*}
      In particular, Thus, $U\left( f,\rho_1,\delta \right)\subseteq \operatorname{ev}_0^{-1}\left( U\left( z,\ve \right) \right)$, so that $\operatorname{ev}_0^{-1}\left( U\left( z,\ve \right) \right)$ is open as any element in this set admits a basic open neighborhood contained in the set. Therefore, $\operatorname{ev}_0$ is continuous.

      Since $V_0 = \operatorname{ev}_0^{-1}\left( \set{0} \right)$, it follows that $V_0$ is closed since $\set{0}\subseteq \R$ is closed.
    \end{proof}
    We claim that the subspace topology on $V_0$ is given by $\mathcal{T}_0 = \mathcal{T}\left( \set{\rho_k\vert_{V_0} : k\geq 0} \right)$.
    \begin{proof}
      First, we will show that, given a subbasic open neighborhood $U = V_0\cap U\left( x,\rho_k,\ve \right)$ in the subspace topology of $V_0$ and any element $y$ in this subbasic open neighborhood, there is an open neighborhood $V\in \mathcal{T}\left( \set{\rho_k\vert_{V_0} : k\geq 0} \right)$ such that such that $y\in V\subseteq U$.

      If $y\in U$, we observe that $y\in U\left( x,\rho_k,\ve \right)$. In particular, since these form a subbase for $\mathcal{T}\left( \set{\rho_k : k\geq 0} \right)$, it follows that there is some finite collection $F = \set{\rho_{i_j} : j=1,\dots,m}$ of seminorms and some $\delta > 0$ such that 
      \begin{align*}
        U\left( y,F,\delta \right) &= \bigcap_{j=1}^{m} U\left( y,\rho_{i_j},\delta \right).
      \end{align*}
      Since $U\left( y,\rho_{i_j}|_{V_0},\delta \right)\subseteq U\left( y,\rho_{i_j},\delta \right)$ for each $j$, it follows that we have
      \begin{align*}
        \bigcap_{j=1}^{m} U\left( y,\rho_{i_j}|_{V_0},\delta \right) &\subseteq V_0 \cap \left( \bigcap_{j=1}^{m}U\left( y,\rho_{i_j},\delta \right) \right),
      \end{align*}
      whence setting $V$ to equal the left hand side yields the desired result. Therefore, $ \mathcal{T}\left( \set{\rho_{k}\vert_{V_0} : k\geq 0} \right) $ refines the subspace topology $ \mathcal{T}_0 $.

      In the reverse direction, observe that if $y\in U\left( x,\rho_j|_{V_0},\ve \right)$ in $V_0$, then $y\in U\left( x,\rho_j,\ve \right)$, so there is some $\delta > 0$ and finite $F = \set{\rho_{i_j} : j=1,\dots,m}$ such that 
      \begin{align*}
        U\left( y,F,\ve \right) &= \bigcap_{j=1}^{m} U\left( y,\rho_{i_j},\delta \right)\\
                                &\subseteq U\left( x,\rho_j,\ve \right),
      \end{align*}
      but we notice then that $U\left( x,\rho_j,\ve \right)\cap V_0\subseteq U\left( x,\rho_j|_{V_0},\ve \right)$, so that $V_0\cap U\left( y,F,\ve \right)\subseteq U\left( x,\rho_j|_{V_0},\ve \right)$, so that $ \mathcal{T}_0 $ refines $ \mathcal{T}\left( \set{\rho_k\vert_{V_0} : k\geq 0} \right) $. Thus, these topologies are equal.
    \end{proof}
    We may consider $ \mathcal{B}\left( \mathcal{T}_0 \right) $ to be the Borel $\sigma$-algebra on $V_0$. Brownian motions are a special class of probability measure on $ \left( V_0,\mathcal{B}\left( \mathcal{T}_0 \right) \right) $. Specifically, they are the probability measures such that for every $n\geq 0$, partition $ 0< t_1 < t_2 < \cdots < t_n $, the random variables
    \begin{align*}
      X_1 &= f\mapsto f\left( t_1 \right)\\
      X_2 &= f\mapsto \left( f\left( t_2 \right)-f\left( t_1 \right) \right)\\
      X_3 &= f\mapsto \left( f\left( t_3 \right)-f\left( t_2 \right) \right)\\
          &\vdots\\
      X_n &= f\mapsto \left( f\left( t_n \right)-f\left( t_{n-1} \right) \right)
    \end{align*}
    are independent centered Gaussian random variables with variances $t_1,t_2-t_2,\dots, t_n-t_{n-1}$ for an arbitrary $f$. 

    Similar to the case of $s\left( \N,\K \right)$, the space $\left( V_0,\mathcal{T}_0 \right)$ is not a normed space.
  \item Consider the vector space $V = \R^{\N}$, which is the space of all real-valued sequences. Let $ \mathcal{T} $ be the product topology on $\R^{\N}$. We are curious as to whether $\left( V,\mathcal{T} \right)$ is a locally convex topological vector space.

    Consider the function $d\colon V\times V \rightarrow [0,\infty)$ defined by
    \begin{align*}
      d\left( x,y \right) &= \sum_{n=0}^{\infty} 2^{-n} \min\left( 1, \left\vert x_n-y_n \right\vert \right).
    \end{align*}
    Then, we have that $d$ is a distance metric, and $ \mathcal{T} = \mathcal{T}\left( d \right) $.
    \begin{proof}
      We start by showing that if $x\in V$ is any element and $\ve > 0$, then $U\left( x,\ve \right)$ is open in $ \mathcal{T} $. Translating, we take $x = 0$ (since the metric $d$ is translation-invariant), so that we are now seeking to show that $U\left( 0,\ve \right)$ is open. This is equivalent to showing that if $x\in U\left( 0,\ve \right)$, then there is a finite collection of subbasic open sets $\pi_j^{-1}\left( U_j \right)$ whose intersection $W$ contains $x$ and is contained in $U\left( 0,\ve \right)$.

      To start, we observe that there is some index $n_0$ such that
      \begin{align*}
        \sum_{n=n_0 + 1}^{\infty} 2^{-n} &< \ve/2.
      \end{align*}
      We then select $U_j = \left( x_j-\ve/4,x_j+\ve/4 \right)$ for each $j=0,\dots,n_0$. It then follows that $x\in W$, and we have for any $y\in W$ that
      \begin{align*}
        d\left( x,y \right) &= \sum_{j=0}^{\infty} 2^{-j}\min\left( 1,\left\vert x_j-y_j \right\vert \right)\\
                            &\leq \sum_{j=0}^{n_0} 2^{-j}\min\left( 1,\left\vert x_j-y_j \right\vert \right) + \sum_{j=n_0 + 1}^{\infty} 2^{-j}\\
                            &< \sum_{j=0}^{n_0} 2^{-j} \left( \ve/4 \right) + \ve/2\\
                            &\leq \ve,
      \end{align*}
      whence $x\in W\subseteq U\left( 0,\ve \right)$.

      Now, consider a basic open set $W = \bigcap_{j=0}^{n_0}\pi_j^{-1}\left( U_j \right)$. Let $x\in W$. Let $\delta_0,\dots,\delta_{n_0}$ be such that $\left( x_j-\delta_j,x_j + \delta_j \right)\subseteq U_j$ for each $j=1,\dots,n$, and set $\delta = 2^{-n_1}\min\left( \delta_0,\dots,\delta_{n_0} \right)$, where $n_1$ is any integer strictly larger than $n_0$.

      Notice then that if $y\in U\left( x,\delta \right)$, then we have for each $j=0,\dots,n_0$ that
      \begin{align*}
        \left\vert x_j - y_j \right\vert &< 2^{j}\delta\\
                                         &\leq 2^{n_0}2^{-n_1}\min\left( \delta_0,\dots,\delta_{n_0} \right)\\
                                         &< \min\left( \delta_0,\dots,\delta_{n_0} \right)\\
                                         &\leq \delta_j.
      \end{align*}
      In particular, this means we must have $U\left( x,\delta \right) \subseteq W$.

      Therefore, we have that $\mathcal{T}$ and $\mathcal{T}\left( d \right)$ generate the same topology on $V$.
    \end{proof}
    In particular, we may consider the seminorms
    \begin{align*}
      \rho_n(x) &= \left\vert x_n \right\vert,
    \end{align*}
    and $ \mathcal{N} = \set{\rho_n : n\geq 0} $. Then, we claim that $ \mathcal{T}= \mathcal{T}\left( \mathcal{N} \right) $. 
    \begin{proof}
      A basic open neighborhood of $x\in V$ in $ \mathcal{T}\left( \mathcal{N} \right) $ is given by the sets
      \begin{align*}
        U\left( x,F,\ve \right) &= \bigcap_{j\in F} \set{\left( y_n \right)_n : \left\vert x_j - y_j \right\vert < \ve}.
      \end{align*}
      Meanwhile, a basic open neighborhood of $x$ in $ \mathcal{T} $ is given by the sets
      \begin{align*}
        U'\left( x,F,\ve \right) &= \bigcap_{j\in F} \pi_j^{-1} \left( \left( x_j-\ve,x_j+\ve \right) \right).
      \end{align*}
      In fact, these are the same subsets of $V$ by the definition of the projection map $\pi_j\colon V\rightarrow \R$ taking $\left( x_n \right)_n\mapsto x_j$.

      Therefore, if we have $y\in U\left( x,F,\ve \right)$ in $\mathcal{T}\left( \mathcal{N} \right)$, then if we define 
      \begin{align*}
        \delta &= \frac{1}{2}\min\left( \set{\left\vert \ve-y_j \right\vert : y_j\in F} \right),
      \end{align*}
      we have that $U'\left( y,F,\delta \right) \subseteq U\left( x,F,\ve \right)$. Similarly, if $y\in U'\left( x,F,\ve \right)$, then setting $\delta$ to equal the same as above, we have $U\left( y,F,\ve \right)\subseteq U'\left( x,F,\ve \right)$.

      This gives $\mathcal{T} = \mathcal{T}\left( \mathcal{N} \right)$.
    \end{proof}
    In particular, $V$ is not a normed space.
  \item Consider $c_{00}\left( \N,\K \right)\cong \K[x]$ to consist of all the finitely supported sequences $\left( x_n \right)_{n\geq 0}$. This is a vector space. Consider the seminorms $ \mathcal{N} = \operatorname{SN}\left( V \right) $, and the locally convex space $\left( V, \mathcal{T}\left( \mathcal{N} \right) \right)$. Then, this is a locally convex space that is not metrizable. The topology $ \mathcal{T}\left( \mathcal{N} \right) $ is known as the finest locally convex topology.
\end{enumerate}
\subsection{General Properties of Topological Vector Spaces}%
\begin{definition}
  Let $\left( V,\mathcal{T} \right)$ be a $\K$-TVS, and let $\tau\in \operatorname{SN}\left( V \right)$. Then, $\tau$ is a \textit{continuous seminorm} for $\left( V, \mathcal{T} \right)$ if $\tau\colon V \rightarrow [0,\infty)\subseteq \R$ is continuous.
\end{definition}
\begin{proposition}
  Let $V$ be a $\K$-vector space, $ \mathcal{N}\subseteq \operatorname{SN}\left( V \right) $ define a locally convex topology on $V$. If $\tau\in \operatorname{SN}\left( V \right)$, then $\tau$ is continuous for $ \left( V, \mathcal{T}\left( \mathcal{N} \right) \right) $ if and only if there exists $C\geq 0$, $n\in \N$, and $\rho_1,\dots,\rho_n\in \mathcal{N}$ such that
  \begin{align*}
    \tau\left( v \right) &\leq C \max\set{\rho_j(v) : j=1,\dots,n}\label{eq:continuous_seminorms}\tag{$\ast$}
      \intertext{or}
    \tau\left( v \right) &\leq C \sum_{j=1}^{n} \rho_j(v).
  \end{align*}
\end{proposition}
\begin{proof}
  Suppose $\tau$ is continuous. Then, $[0,1)$ is open in $[0,\infty)$, so $\tau^{-1}\left( [0,1) \right)$ is open in $ \mathcal{T}\left( \mathcal{N} \right) $ with $0\in \tau^{-1}\left( [0,1) \right)$.

  So, by definition, there exists $\ve > 0$ and $\rho_1,\dots,\rho_n$ such that
  \begin{align*}
    \bigcap_{j=1}^{n} U\left( 0,\rho_j,\ve \right) &\subseteq \tau^{-1}\left( [0,\infty) \right).
  \end{align*}
  That is, for any $i\in [n]$, we have that if $\rho_i(x) < \ve$, then $\tau(x) < 1$.

  Let $x\in V$ and $\lambda > 0$. Then,
  \begin{align*}
    \tau(x) &= \tau\left( \lambda \lambda^{-1}x \right)\\
            &= \lambda \tau\left( \lambda^{-1}x \right)\\
            &< \lambda
  \end{align*}
  if for every $i$, we have $\rho_i\left( \lambda^{-1}x \right) < \ve$. 

  Define $J = \set{\lambda > 0 : \rho_i\left(\lambda^{-1}x\right) < \ve\text{ for all }i\in [n]}$. Rewriting, we have
  \begin{align*}
    J &= \set{\lambda > 0 : \frac{1}{\ve}\rho_i\left( x \right) < \lambda\text{ for all }i\in [n]}\\
      &= \bigcap_{i=1}^{n} \left( \frac{1}{\ve}\rho_i(x),\infty \right),
  \end{align*}
  giving $\inf(J) = \frac{1}{\ve}\max\set{\rho_i(x) : 1\leq i \leq n}$. Therefore, if $\lambda\in J$, then $\tau(x) \leq \lambda$, whence $\tau(x) \leq \inf(J)$. Therefore, we have shown that $\tau(x) \leq \frac{1}{\ve}\max\set{\rho_i(x) : 1\leq i \leq n}$.

  In the reverse direction, suppose \eqref{eq:continuous_seminorms} holds. Let $x\in V$, $\rho_1,\dots,\rho_n\in \mathcal{N}$, and let $\eta > 0$. Let
  \begin{align*}
    y &\in \bigcap_{i=1}^{n} U\left( x,\rho_i,\eta \right).
  \end{align*}
  By the reverse triangle inequality, we have
  \begin{align*}
    \left\vert \tau\left( y \right) - \tau\left( x \right) \right\vert &\leq \tau\left( y-x \right)\\
                                                                       &< \ve,
  \end{align*}
  but
  \begin{align*}
    \tau\left( y-x \right) &\leq C \max\set{\rho_i(y-x) : 1\leq i \leq n}\\
                           &\leq C \eta\\
                           &< \ve
  \end{align*}
  if $\eta$ is small enough. Thus, we have for all $y\in \bigcap_{j=1}^{n}U\left( x,\rho_j,\eta \right)$, then $\left\vert \tau(y)-\tau(x) \right\vert < \ve$, whence $\tau$ is continuous at $x$.
\end{proof}
\begin{remark}
  If $\tau\in \mathcal{N}$, then $\tau$ is continuous with respect to $ \mathcal{T}\left( \mathcal{N} \right) $.
\end{remark}
\begin{proposition}
  Let $\left( V, \mathcal{T}\left( \mathcal{N} \right) \right)$ be a LCTVS. Then, $U\subseteq V$ is open if and only if for all $x\in U$, there is some continuous seminorm $\tau$ and some $\ve > 0$ such that $U\left( x,\tau,\ve \right)\subseteq U$.
\end{proposition}
\begin{theorem}[Continuity of Multilinear Maps on LCTVSes]
  Suppose $n\geq 1$, and $V_1,\dots,V_n,W$ are $\K$-LCTVSes, and let $M\colon V_1\times\cdots\times V_n\rightarrow W$ be a $\K$-multilinear map.

  Then, $M$ is (jointly) continuous if and only if for all continuous seminorm $\tau$ on $W$, there exist $\rho_1,\dots,\rho_n$ continuous seminorms on $V_1,\dots,V_n$ respectively such that for all tuples $\left( x_1,\dots,x_n \right)\in V_1\times\cdots\times V_n$,
  \begin{align*}
    \tau\left( M\left( x_1,\dots,x_n \right) \right) &\leq \rho\left( x_1 \right)\cdots\rho\left( x_n \right).\label{eq:continuous_multilinear_map}\tag{$\dag$}
  \end{align*}
\end{theorem}
\begin{proof}
  Suppose $M$ is (jointly) continuous. That is, $M$ is equipped with the product topology. Since $\tau$ is a continuous seminorm, we have that $\tau$ is a continuous map from $W \rightarrow [0,\infty)$, we have $\tau^{-1}\left( [0,1) \right)$ is open in $W$. Moreover, $M\left( 0,\dots,0 \right) = 0\in \tau^{-1}\left( [0,1) \right)$. Since $M$ is continuous at $\left( 0,\dots,0 \right)$, there is a neighborhood $U_1\times \dots\times U_n$ in $V_1\times\cdots\times V_n$ where each $U_i$ is a neighborhood of $0$ in $V_i$, such that $M\left( U_1\times\cdots\times U_n \right)\subseteq \tau^{-1}\left( [0,1) \right)$.

  Let $\left( x_1,\dots,x_n \right)\in V_1\times\cdots\times V_n$, and let $\lambda_1,\dots,\lambda_n > 0$. We have
  \begin{align*}
    \tau\left( M\left( x_1,\dots,x_n \right) \right) &= \tau\left( M\left( \lambda_1\lambda_1^{-1}x_1,\dots,\lambda_n\lambda_n^{-1}x_n \right) \right)\\
                                                     &= \lambda_1\cdots\lambda_n \tau\left( M\left( \lambda_1^{-1}x_1,\dots,\lambda_n^{-1}x_n \right) \right)\\
                                                     &< \lambda_1\cdots\lambda_n
  \end{align*}
  if for all $i$, we have $\lambda_i^{-1}x_i\in U_i$. For each $i\in [n]$, we have $U_i$ is an open neighborhood of $0\in V_i$, so there exists some continuous seminorm $\sigma_i$ and $\ve_i > 0$ such that $U\left( 0,\sigma_i,\ve_i \right)\subseteq U_i$. If for every $i$, we have $\sigma_i\left( \lambda_i^{-1}x_i \right) < \ve_i$, then
  \begin{align*}
    \tau\left( M\left( x_1,\dots,x_n \right) \right) &< \lambda_1\cdots\lambda_n.
  \end{align*}
  Rewriting, we have $\lambda_i > \frac{1}{\ve_i} \sigma_i\left(x_i\right)$ for each such $i$, so that we have
  \begin{align*}
    \tau\left( M\left( x_1,\dots,x_n \right) \right) &\leq \lambda_1\cdots\lambda_n\\
                                                     &\leq \inf\left( \set{\lambda_1\cdots\lambda_n : \lambda_i > 0\text{ and for all $i\in [n]$, }\lambda_i > \frac{1}{\ve_i}\sigma_i\left( x_i \right)} \right)\\
                                                     &= \left( \frac{1}{\ve_1}\sigma_1\left(x_1\right) \right)\cdots \left( \frac{1}{\ve_n}\sigma_n\left( x_n \right) \right)
  \end{align*}
  Setting $\rho_i = \frac{1}{\ve_i}\sigma_i$ yields the desired result.

  Assume that \eqref{eq:continuous_multilinear_map} holds. Let $\left( x_1,\dots,x_n \right),\left( y_1,\dots,y_n \right)\in V_1\times\cdots\times V_n$. We want to try to compare the difference between $M\left( x_1,\dots,x_n \right)$ and $M\left( y_1,\dots,y_n \right)$. From multilinearity, we have
  \begin{align*}
    M\left( x_1,\dots,x_n \right) - M\left( y_1,\dots,y_n \right)  &=  \sum_{i=1}^{n} \left( M\left( y_1,\dots,y_{i-1},y_i,x_{i+1},\dots, x_n\right) - M\left( y_1,\dots,y_{i-1},x_i,x_{i+1},\dots,x_n \right) \right)\\
                                                                   &= \sum_{i=1}^{n} M\left( y_1,\dots,y_{i-1},y_i-x_i,x_{i+1},\dots,x_n \right)
  \end{align*}
  Let $Z\subseteq W$ be open containing $z = M\left( x_1,\dots,x_n \right)$, and write $z' = M\left( y_1,\dots,y_n \right)$. Then, there exists a continuous seminorm $\tau$ on $W$ and $\ve > 0$ such that $U\left( z,\tau,\ve \right)\subseteq Z$.

  Consider $\tau\left( z-z' \right)$. Then, we have by multilinearity and the triangle inequality that
  \begin{align*}
    \tau\left( z-z' \right) &\leq \sum_{i=1}^{n} \tau\left( M\left( y_1,\dots,y_{i-1},y_i-x_i,x_{i+1},\dots,x_n \right) \right).
    \intertext{Using \eqref{eq:continuous_multilinear_map} gives continuous seminorms $\rho_1,\dots,\rho_n$ on $V_1,\dots,V_n$ respectively such that}
                            &\leq \sum_{i=1}^{n} \rho_1\left( y_1 \right)\cdots \rho_{i-1}\left( y_{i-1} \right)\rho_i\left( y_i-x_i \right)\rho_{i+1}\left( x_{i+1} \right)\cdots \rho_n\left( x_n \right)\\
                            &\leq n \left( C + \eta \right)^{n-1}\eta,
  \end{align*}
  where we assume that for all $i$, $y_i\in U\left( x_i,\rho_i,\eta \right)$, and $C = \max\set{\rho_i\left( x_i \right) : i=1,\dots,n}$. By choosing $\eta$ sufficiently small, $n\left( C + \eta \right)^{n-1}\eta < \ve$, which yields 
  \begin{align*}
    M\left( \prod_{i=1}^{n}U\left( x_i,\rho_i,\eta \right) \right) &\subseteq U\left( z,\tau,\ve \right)\\
                                                                   &\subseteq Z,
  \end{align*}
  so $M$ is continuous.
\end{proof}
\section{Notation}%
\begin{itemize}
  \item The natural numbers are given by $\N = \set{0,1,2,3,\dots}$.
  \item The strictly positive natural numbers are given by $\Z_{> 0} = \set{1,2,3,\dots}$.
  \item A multi-index is a tuple $\alpha\in \N^{n}$. We let
    \begin{align*}
      \left\vert \alpha \right\vert = \alpha_1 + \cdots + \alpha_n.
    \end{align*}
    If $x = \left( x_1,\dots,x_n \right)\in \R^{n}$, then we will write 
    \begin{align*}
      x^{\alpha} &= x_1^{\alpha_1}\cdots x_n^{\alpha_n}
        \intertext{and}
      \alpha! &= \alpha_1!\cdots\alpha_n!
    \end{align*}
    for the corresponding values.

    If $f\colon \R^{n}\rightarrow \R$ is any function, then we will write
    \begin{align*}
      \partial^{\alpha}f &= \frac{\partial^{\left\vert \alpha \right\vert}}{\partial x_1^{\alpha_1}\cdots \partial x_n^{\alpha_n}}.
    \end{align*}
  \item If $A$ is a set, we will write $ \mathcal{P}(A) $ for the power set of $A$. If $A$ and $B$ are sets, then the set $B^{A}\subseteq \mathcal{P}\left( A\times B \right)$ is the set of all functions $f\colon A\rightarrow B$.

    We will write $ \mathcal{P}_{\operatorname{fin}}\left( X \right) $ to consist of all finite subsets subsets of $X$.
  \item If $x\in \R^{n}$ is given by $\left( x_1,\dots,x_n \right)$, we will define the bracket 
    \begin{align*}
      \left\langle x \right\rangle \coloneq  \sqrt{1 + x_1^2 + \cdots + x_n^2}.
    \end{align*}
    In the particular case of $n = 1$, we will write $ \left\langle k \right\rangle = \sqrt{1 + k^2} $ for any $k\in \N$.
  \item If $n\in \N$, then
    \begin{align*}
      \left[ n \right] &= \set{1,2,\dots,n},
    \end{align*}
    where we have $\left[ 0 \right] = \emptyset$.
  \item If $A$ is a finite set, then $\left\vert A \right\vert$ is the cardinality of $A$. 
  \item Generally we will denote by $ \K $ for either $\R$ or $\C$.
\end{itemize}
